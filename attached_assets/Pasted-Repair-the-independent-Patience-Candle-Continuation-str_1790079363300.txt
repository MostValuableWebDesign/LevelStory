const strategyId = canonicalStrategyId(evaluation.setupType);

Recommended implementation shape
Create or use a small helper with behavior equivalent to:
function isEvaluationEnabled(
  setupType: string,
  enabledStrategies: VisualReviewStrategyToggles | undefined,
): boolean {
  if (!enabledStrategies) return true;

  if (setupType === "PATIENCE_CANDLE_CONTINUATION") {
    return enabledStrategies.PATIENCE_CANDLE_CONTINUATION !== false;
  }

  const canonical = canonicalStrategyId(setupType);
  return canonical === null || enabledStrategies[canonical] !== false;
}
Use this helper when filtering snapshot.setupAnalysis.evaluations.
